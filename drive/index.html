// Replace your existing window.onload and auth logic with this const API_BASE = "https://simple-liana-techzone3201-048a28fa.koyeb.app"; function login() { window.location.href = API_BASE + "/auth/login"; } function handleAuthData() { // Check if we just arrived from a successful login const hash = window.location.hash; if (hash.includes('auth_data=')) { try { const rawData = hash.split('auth_data=')[1]; const decodedData = decodeURIComponent(rawData); // Save the key to techzonex.store's storage localStorage.setItem('drive_creds', decodedData); // Clean the URL so the key doesn't stay in history window.history.replaceState(null, null, window.location.pathname); console.log("✅ Auth data captured successfully"); } catch (e) { console.error("❌ Failed to parse auth data", e); } } // Update the UI const creds = localStorage.getItem('drive_creds'); if (creds) { document.getElementById('auth-status').innerHTML = `
Authorized
`; } } // Ensure this runs when the page loads window.onload = handleAuthData;
